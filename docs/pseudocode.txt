CONSTANTS
    Declare MAX_SEARCH_RADIUS_KM As Integer = 25
    Declare RADIUS_STEP_KM As Integer = 5
    Declare DISRUPTION_KEYWORDS As String() = ["flooding", "school closed",
        "teacher strike", "no tutor present"]

DATA TYPES
    Student
        studentID   : Integer
        name        : String
        age         : Integer
        phone       : String
        location    : String

    LearningCenter
        centerID        : Integer
        name            : String
        location        : String
        tutorOnDuty     : Boolean
        tutorSubject    : String
        openSeats       : Integer
        materialStock   : List of MaterialRecord

    MaterialRecord
        materialName : String
        quantity     : Integer

    Session
        sessionID : Integer
        studentID : Integer
        centerID  : Integer
        date      : Date
        status    : String

-----------------------------------------------------------
Function mainMenu()
-----------------------------------------------------------
    Display "1. Check Tutor Availability"
    Display "2. Check Study Material Stock"
    Display "3. Book Tutoring Session"
    Display "4. Report School Disruption"
    Input userChoice
    Call routeRequest(userChoice)
End Function

-----------------------------------------------------------
Function routeRequest(choice)
-----------------------------------------------------------
    If choice = 1 Then
        Call checkTutorAvailability()
    Else If choice = 2 Then
        Call checkMaterialStock()
    Else If choice = 3 Then
        Call bookTutoringSession()
    Else If choice = 4 Then
        Call reportDisruption()
    Else
        Display "Invalid option, please try again"
        Call mainMenu()
    End If
End Function

-----------------------------------------------------------
Function checkTutorAvailability()
-----------------------------------------------------------
    Input studentLocation
    Input subjectNeeded
    Set radius = RADIUS_STEP_KM
    Set found = false

    While (found = false) And (radius <= MAX_SEARCH_RADIUS_KM)
        Set nearbyCenters = findCentersWithin(studentLocation, radius)
        For Each center In nearbyCenters
            If center.tutorOnDuty = true And center.tutorSubject = subjectNeeded
               And center.openSeats > 0 Then
                Display center.name & " - " & center.openSeats & " seat(s) available"
                Set found = true
            End If
        End For
        If found = false Then
            Set radius = radius + RADIUS_STEP_KM
        End If
    End While

    If found = false Then
        Display "No tutor availability found within " & MAX_SEARCH_RADIUS_KM & " km"
    End If
    Call logInteraction()
End Function

-----------------------------------------------------------
Function checkMaterialStock()
-----------------------------------------------------------
    Input studentLocation
    Input materialName
    Set nearbyCenters = findCentersWithin(studentLocation, MAX_SEARCH_RADIUS_KM)
    Set resultsFound = false

    For Each center In nearbyCenters
        For Each record In center.materialStock
            If record.materialName = materialName And record.quantity > 0 Then
                Display center.name & " has " & record.quantity & " copies of " & materialName
                Set resultsFound = true
            End If
        End For
    End For

    If resultsFound = false Then
        Display materialName & " not currently in stock nearby"
    End If
    Call logInteraction()
End Function

-----------------------------------------------------------
Function bookTutoringSession()
-----------------------------------------------------------
    Input studentPhone
    Set student = findStudentByPhone(studentPhone)

    If student = null Then
        Call registerStudent()
        Set student = findStudentByPhone(studentPhone)
    End If

    Input preferredCenterID, preferredDate
    Set center = findCenterByID(preferredCenterID)

    If center.openSeats > 0 Then
        Set newSession.sessionID = generateID()
        Set newSession.studentID = student.studentID
        Set newSession.centerID = center.centerID
        Set newSession.date = preferredDate
        Set newSession.status = "Confirmed"
        Set center.openSeats = center.openSeats - 1
        Call sendNotification(student.phone,
             "Tutoring session confirmed at " & center.name & " on " & preferredDate)
    Else
        Display "No seats available on that date. Suggesting next available date..."
        Call suggestNextAvailable(center)
    End If
    Call logInteraction()
End Function

-----------------------------------------------------------
Function reportDisruption()
-----------------------------------------------------------
    Input studentLocation, disruptionDescription
    Set isDisruption = false

    For Each keyword In DISRUPTION_KEYWORDS
        If disruptionDescription Contains keyword Then
            Set isDisruption = true
        End If
    End For

    If isDisruption = true Then
        Set radius = RADIUS_STEP_KM
        Set found = false

        While (found = false) And (radius <= MAX_SEARCH_RADIUS_KM)
            Set nearbyCenters = findCentersWithin(studentLocation, radius)
            For Each center In nearbyCenters
                If center.tutorOnDuty = true Then
                    Display "REDIRECT: Nearest open center is " & center.name
                    Call sendNotification(center.phone,
                         "Incoming redirected student from " & studentLocation)
                    Set found = true
                End If
            End For
            If found = false Then
                Set radius = radius + RADIUS_STEP_KM
            End If
        End While

        If found = false Then
            Display "No open learning center found. Contact District Education Office."
        End If
    Else
        Display "No disruption keyword detected. Redirecting to Book Tutoring Session."
        Call bookTutoringSession()
    End If
    Call logInteraction()
End Function

-----------------------------------------------------------
Function registerStudent()
-----------------------------------------------------------
    Input name, age, phone, location
    Set newStudent.studentID = generateID()
    Set newStudent.name = name
    Set newStudent.age = age
    Set newStudent.phone = phone
    Set newStudent.location = location
    Store newStudent
    Display "Registration successful"
End Function

-----------------------------------------------------------
Function sendNotification(phone, message)
-----------------------------------------------------------
    Send SMS To phone With message
End Function

-----------------------------------------------------------
Function logInteraction()
-----------------------------------------------------------
    Store timestamp, userChoice, outcome In interactionLog
    (used later for District Education Office reporting and resource planning)
End Function

-----------------------------------------------------------
Main
-----------------------------------------------------------
Program
    Call mainMenu()
End Program
